package routing

import (
	"slices"
	"testing"
)

func TestCorpus(t *testing.T) {
	want := []RiskDomain{RiskSecurity, RiskAuthentication, RiskAuthorization, RiskSecrets, RiskCryptography, RiskMigrations, RiskConcurrency, RiskDataIntegrity, RiskDestructiveOperations}
	t.Run("closed", func(t *testing.T) {
		for _, d := range want {
			if !d.Valid() {
				t.Errorf("known domain rejected: %s", d)
			}
		}
		for _, d := range []RiskDomain{"", "unknown", "Security"} {
			if d.Valid() {
				t.Errorf("unknown domain accepted: %q", d)
			}
		}
	})
	t.Run("order", func(t *testing.T) {
		if got := Domains(); !slices.Equal(got, want) {
			t.Fatalf("Domains = %v, want %v", got, want)
		}
	})
	t.Run("copy", func(t *testing.T) {
		before := slices.Clone(Domains())
		got := Domains()
		got[0] = "mutated"
		if !slices.Equal(Domains(), before) {
			t.Fatal("caller mutated later domain results")
		}
	})
}
